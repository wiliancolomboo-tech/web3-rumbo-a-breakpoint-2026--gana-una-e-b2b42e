FROM node:20-slim AS runner

WORKDIR /app

RUN apt-get update && apt-get install -y --no-install-recommends \
    curl \
    ca-certificates \
    build-essential \
    && rm -rf /var/lib/apt/lists/*

COPY package*.json ./
COPY tsconfig.json ./
RUN npm ci

COPY client ./client

ENV NODE_ENV=production
ENV PORT=8080

EXPOSE 8080

CMD ["npx", "ts-node", "client/worker.ts"]
