import express from "express";
import { evaluate } from "../../../packages/engine/src/index.js";
import { sendWinnerEmail } from "./resend.js";
import { notifyWebhook } from "./webhook.js";

const app = express();
app.use(express.json({ limit: "64kb" }));

const decisions = new Map<string, ReturnType<typeof evaluate>>();

app.get("/healthz", (_req, res) => {
  res.json({ status: "ok", service: "rumbo-proof", uptime: process.uptime() });
});

app.post("/v1/campaigns/:campaignId/applications", async (req, res) => {
  try {
    const { campaignId } = req.params;
    const { wallet, email, tasks } = req.body;

    const decision = evaluate(
      { wallet, email, tasks },
      Number(process.env.MIN_SCORE ?? 70),
      campaignId
    );

    decisions.set(decision.candidateHash, decision);

    await notifyWebhook({
      event: decision.eligible ? "candidate.eligible" : "candidate.ineligible",
      campaignId,
      wallet: decision.wallet,
      score: decision.score,
      candidateHash: decision.candidateHash,
    });

    res.status(201).json({
      ...decision,
      email: undefined,
    });
  } catch (error) {
    res.status(400).json({
      error: error instanceof Error ? error.message : "invalid_request",
    });
  }
});

app.post("/v1/admin/notify-winner", async (req, res) => {
  const { email, wallet, txSignature } = req.body;

  if (
    typeof email !== "string" ||
    !email.includes("@") ||
    typeof wallet !== "string" ||
    typeof txSignature !== "string"
  ) {
    return res.status(400).json({ error: "invalid_payload" });
  }

  await sendWinnerEmail({ email, wallet, txSignature });
  res.status(202).json({ accepted: true });
});

const port = Number(process.env.PORT ?? 3000);
app.listen(port, () => console.log(`RumboProof listening on ${port}`));
