FROM node:20-slim AS builder

WORKDIR /usr/src/app

COPY package*.json ./
COPY tsconfig.json ./
RUN npm ci

COPY client/ ./client/
RUN npm run build || npx tsc client/orchestrator.ts --outDir dist/

FROM node:20-slim AS runner

WORKDIR /usr/src/app
ENV NODE_ENV=production

COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

USER node
EXPOSE 8080

CMD ["node", "dist/orchestrator.js"]
